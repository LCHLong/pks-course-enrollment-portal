server/src/app.ts Clean Architecture Express Gateway 
import express, { Application, Request, Response } from 'express';
import cors from 'cors';
import { apiRouter } from './routes/index.js';
import { errorHandler } from './middlewares/error.middleware.js';

export const createApp = (): Application => {
  const app = express();

  // 1. CORS Configuration cho phép Vercel Client & Localhost
  const allowedClient = process.env.CLIENT_URL;
  app.use(cors({
    origin: (origin, callback) => {
      if (!origin) return callback(null, true);
      if (
        !allowedClient ||
        allowedClient === '*' ||
        origin === allowedClient ||
        origin.endsWith('.vercel.app') ||
        origin.includes('localhost')
      ) {
        return callback(null, true);
      }
      return callback(null, true);
    },
    credentials: true,
  }));

  // 2. Body Parser Middlewares
  app.use(express.json());
  app.use(express.urlencoded({ extended: true }));

  // 3. Healthcheck Endpoint đồng bộ Múi giờ Asia/Ho_Chi_Minh
  app.get(['/', '/health', '/api/health'], (_req: Request, res: Response) => {
    res.json({
      status: 'ok',
      service: 'PKS Course & Enrollment Portal API',
      timestamp: new Date().toISOString(),
      timezone: 'Asia/Ho_Chi_Minh',
    });
  });

  // 4. RESTful API Routes (/api/auth, /api/courses, /api/student, /api/admin)
  app.use('/api', apiRouter);

  // 5. Global Error Handler Middleware
  app.use(errorHandler);

  return app;
};

3. Sơ Đồ Quan Hệ Thực Thể Chi Tiết (Entity Relationship Diagram - ERD)

Chuẩn hóa 3NF với 3 thực thể users, courses, enrollments, ràng buộc toàn vẹn khóa ngoại onDelete: Restrict và compound unique (user_id, course_id).

Prisma 3NF Relational Model
server/prisma/schema.prisma Database Schema Contract 
datasource db {
  provider  = "postgresql"
  url       = env("DATABASE_URL")
  directUrl = env("DIRECT_URL")
}

generator client {
  provider = "prisma-client-js"
}

enum Role {
  student
  admin
}

enum EnrollmentStatus {
  enrolled
  confirmed
  cancelled
  pending
}

model User {
  id            String       @id @default(uuid())
  name          String
  email         String       @unique
  password_hash String
  role          Role         @default(student)
  created_at    DateTime     @default(now())
  updated_at    DateTime     @updatedAt
  enrollments   Enrollment[]

  @@map("users")
}

model Course {
  id             String       @id @default(uuid())
  title          String
  category       String
  instructor     String
  description    String
  price          Int          // Lưu giá tiền theo VND (nguyên số)
  capacity       Int          // Sĩ số tối đa (> 0)
  enrolled_count Int          @default(0) // Số lượng học viên đã ghi danh
  is_active      Boolean      @default(true)
  created_at     DateTime     @default(now())
  updated_at     DateTime     @updatedAt
  enrollments    Enrollment[]

  @@index([category])
  @@index([is_active])
  @@map("courses")
}

model Enrollment {
  id         String           @id @default(uuid())
  user_id    String
  course_id  String
  status     EnrollmentStatus @default(enrolled)
  created_at DateTime         @default(now())
  updated_at DateTime         @updatedAt

  // Ràng buộc Restrict bảo vệ toàn vẹn lịch sử giao dịch
  user   User   @relation(fields: [user_id], references: [id], onDelete: Restrict)
  course Course @relation(fields: [course_id], references: [id], onDelete: Restrict)

  // Ràng buộc duy nhất: Một học viên chỉ được ghi danh một khóa học đúng 1 lần
  @@unique([user_id, course_id])
  @@map("enrollments")
}

import { Request, Response, NextFunction } from 'express';
import { sendError } from '../utils/apiResponse.js';

export const requireRole = (role: 'student' | 'admin') => {
  return (req: Request, res: Response, next: NextFunction): void => {
    if (!req.user) {
      sendError(res, 'UNAUTHORIZED', 'Chưa xác thực người dùng.', 401);
      return;
    }

    // Chặn đứng Student cố tình gọi API Admin với mã 403 Forbidden
    if (req.user.role !== role) {
      sendError(
        res,
        'FORBIDDEN',
        'Truy cập bị từ chối: Bạn không có quyền truy cập chức năng này.',
        403
      );
      return;
    }

    next();
  };
};
client/src/components/layout/ProtectedRoute.tsx 
import React from 'react';
import { Navigate, useLocation } from 'react-router-dom';
import { toast } from 'sonner';
import { useAuth } from '@/components/ui/AuthContext';

export const ProtectedRoute: React.FC<{
  children: React.ReactNode;
  allowedRoles?: string[];
}> = ({ children, allowedRoles }) => {
  const { user, isAuthenticated, isLoading } = useAuth();
  const location = useLocation();

  if (isLoading) return <div>Đang xác thực...</div>;

  if (!isAuthenticated) {
    const returnUrl = encodeURIComponent(location.pathname + location.search);
    return <Navigate to={`/login?returnUrl=${returnUrl}`} replace />;
  }

  // Chặn học viên truy cập trang quản trị Admin
  if (allowedRoles && allowedRoles.length > 0 && user && !allowedRoles.includes(user.role)) {
    toast.error('Bạn không có quyền truy cập trang quản trị này.');
    return <Navigate to="/courses" replace />;
  }

  return <>{children}</>;
};

5. Sơ Đồ Tuần Tự Xử Lý Race Condition & Concurrency Tranh Suất Cuối

Mô phỏng trường hợp 2 học viên bấm ghi danh tại cùng một mili-giây khi khóa học chỉ còn lại duy nhất 1 suất (capacity = 20, enrolled_count = 19).

Atomic Conditional Update Execution Sequence
server/src/services/enrollment.service.ts Core Atomic Counter Logic 
export class EnrollmentService {
  static async enrollCourse(userId: string, courseId: string) {
    const trimmedCourseId = courseId.trim();

    return await prisma.$transaction(async (tx) => {
      // 1. Kiểm tra khóa học tồn tại & đang mở ghi danh
      const targetCourse = await tx.course.findUnique({
        where: { id: trimmedCourseId },
      });

      if (!targetCourse) {
        throw new AppError('COURSE_NOT_FOUND', 'Không tìm thấy khóa học.', 404);
      }
      if (!targetCourse.is_active) {
        throw new AppError('COURSE_INACTIVE', 'Khóa học hiện không mở đăng ký.', 400);
      }

      // 2. Chặn ghi danh trùng lặp (nếu đã có bản ghi chưa bị hủy)
      const existingEnrollment = await tx.enrollment.findUnique({
        where: {
          user_id_course_id: { user_id: userId, course_id: trimmedCourseId },
        },
      });
      if (existingEnrollment && existingEnrollment.status !== 'cancelled') {
        throw new AppError('ALREADY_ENROLLED', 'Bạn đã ghi danh khóa học này rồi.', 409);
      }

      // 3. Kiểm tra sĩ số trước khi cập nhật
      if (targetCourse.enrolled_count >= targetCourse.capacity) {
        throw new AppError('COURSE_FULL', 'Khóa học đã đủ sĩ số tối đa.', 400);
      }

      // 4. POSTGRESQL ATOMIC CONDITIONAL UPDATE CHỐNG RACE CONDITION
      // Lệnh thực thi nguyên tử ở tầng CSDL: Chỉ tăng count nếu enrolled_count < capacity
      const updateResult = await tx.$executeRaw`
        UPDATE "courses"
        SET "enrolled_count" = "enrolled_count" + 1, "updated_at" = NOW()
        WHERE "id" = ${trimmedCourseId} AND "enrolled_count" < "capacity" AND "is_active" = true
      `;

      // Nếu 0 dòng bị ảnh hưởng (khóa học đã đầy do tiến trình khác vừa chiếm suất cuối)
      if (updateResult === 0) {
        throw new AppError('COURSE_FULL', 'Khóa học đã đủ sĩ số tối đa.', 400);
      }

      // 5. Tạo mới hoặc tái kích hoạt bản ghi enrollment
      let enrollment;
      if (existingEnrollment && existingEnrollment.status === 'cancelled') {
        enrollment = await tx.enrollment.update({
          where: { id: existingEnrollment.id },
          data: { status: 'enrolled', updated_at: new Date() },
          include: { course: true },
        });
      } else {
        enrollment = await tx.enrollment.create({
          data: { user_id: userId, course_id: trimmedCourseId, status: 'enrolled' },
          include: { course: true },
        });
      }

      return {
        ...enrollment,
        registration_date: formatDateVN(enrollment.created_at),
      };
    });
  }
}

6. Sơ Đồ Khóa Dòng Quản Trị & Bảo Vệ Bất Biến Sĩ Số (Capacity Invariant)

Cơ chế Row Locking SELECT ... FOR UPDATE đảm bảo Quản trị viên không bao giờ có thể hạ sĩ số tối đa xuống thấp hơn số học viên đã ghi danh thực tế.

Row-Locking SELECT FOR UPDATE Sequence
server/src/services/admin.service.ts Row-Locking Guard Logic 
export class AdminService {
  static async updateCourse(id: string, data: UpdateCourseDTO) {
    return await prisma.$transaction(async (tx) => {
      // 1. KHÓA DÒNG ĐỘC QUYỀN VỚI SELECT ... FOR UPDATE
      const rows = await tx.$queryRaw<any[]>`
        SELECT * FROM "courses" WHERE "id" = ${id} FOR UPDATE
      `;

      if (!rows || rows.length === 0) {
        throw new AppError('COURSE_NOT_FOUND', 'Không tìm thấy khóa học.', 404);
      }

      const current = rows[0];

      // 2. KIỂM TRA BẤT BIẾN: SĨ SỐ MỚI KHÔNG ĐƯỢC NHỎ HƠN ENROLLED_COUNT THỰC TẾ
      if (data.capacity !== undefined) {
        if (data.capacity <= 0 || data.capacity < current.enrolled_count) {
          throw new AppError(
            'INVALID_CAPACITY',
            `Sĩ số tối đa (${data.capacity}) không hợp lệ hoặc thấp hơn số lượng học viên đã ghi danh thực tế (${current.enrolled_count}).`,
            400
          );
        }
      }

      // 3. Cập nhật khóa học sau khi thỏa mãn bất biến
      const updatedCourse = await tx.course.update({
        where: { id },
        data: {
          title: data.title?.trim(),
          category: data.category?.trim(),
          instructor: data.instructor?.trim(),
          description: data.description?.trim(),
          price: data.price,
          capacity: data.capacity,
          is_active: data.is_active,
        },
      });

      return updatedCourse;
    });
  }
}

Enrollment Finite State Machine
server/src/services/enrollment.service.ts Self-Cancellation & Atomic Decrement 
export async function cancelStudentEnrollment(enrollmentId: string, userId: string) {
  return await prisma.$transaction(async (tx) => {
    const enrollment = await tx.enrollment.findUnique({
      where: { id: enrollmentId },
      include: { course: true },
    });

    if (!enrollment) {
      throw new AppError('NOT_FOUND', 'Không tìm thấy thông tin ghi danh', 404);
    }

    // Kiểm tra quyền sở hữu tránh lỗ hổng IDOR
    if (enrollment.user_id !== userId) {
      throw new AppError('FORBIDDEN', 'Bạn không có quyền hủy khóa học của học viên khác', 403);
    }

    if (enrollment.status === 'cancelled') {
      throw new AppError('ALREADY_CANCELLED', 'Lượt ghi danh này đã được hủy trước đó', 400);
    }

    // 1. Chuyển trạng thái sang cancelled
    const updated = await tx.enrollment.update({
      where: { id: enrollmentId },
      data: { status: 'cancelled' },
      include: { course: true },
    });

    // 2. GIẢM SĨ SỐ NGUYÊN TỬ VỚI CÔNG THỨC GREATEST BẢO VỆ KHÔNG ÂM
    await tx.$executeRaw`
      UPDATE "courses"
      SET "enrolled_count" = GREATEST("enrolled_count" - 1, 0)
      WHERE "id" = ${enrollment.course_id}
    `;

    return {
      ...updated,
      registration_date: formatDateVN(updated.created_at),
    };
  });
}
